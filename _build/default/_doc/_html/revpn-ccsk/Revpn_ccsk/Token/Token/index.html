Module Token.Token

type t =
  1. | Tok_empty
  2. | Tok of Lts.label * t list * int
    (*

    Let N = \langle S, T, F, \lambda, \mathsf{A} \rangle be a labelled net. token is the type of tokens defined inductively as:

    \cfrac{}{\mathsf{Tok\_empty : token}} \qquad\qquad 
     \cfrac{a\in\mathsf{A} \qquad ls\mathsf{ : [token]} \qquad i \in \mathbb N}{\mathsf{Tok}(a, ls, i)\mathsf{ : token}}
    • Tok_empty: (\cdot, \langle\!\langle \rangle\!\rangle, \cdot)
    • Tok (a, ls, i): if ls has token type, (a, \langle\!\langle w_1,\dots,w_n\rangle\!\rangle, i) with a an action, w_1,\dots,w_n are tokens and i\in\mathbb N an identificator number.
    *)
val size : t -> int

The size of a token is defined as follows: \mathsf{size}((\cdot,\langle\!\langle \rangle\!\rangle,\cdot))=0 and \mathsf{size}((a,\langle\!\langle w_1,\dots, w_n \rangle\!\rangle,j))= \max\{\mathsf{size(w_i) \mid 1\le i \le n}\} + 1.
